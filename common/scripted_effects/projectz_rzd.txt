projectz_rzd_maintain_fanaticism = {
	set_war_support = 1
}

projectz_rzd_ensure_horde_template = {
	RZD = {
		if = {
			limit = {
				NOT = { has_template = "Shambling Horde" }
			}
			division_template = {
				name = "Shambling Horde"
				priority = 1
				regiments = {
					projectz_zombie_horde = { x = 0 y = 0 }
				}
			}
		}
	}
}

projectz_rzd_spawn_starting_hordes = {
	event_target:projectz_outbreak_state = {
		create_unit = {
			division = "division_template = \"Shambling Horde\" start_experience_factor = 0.3 start_equipment_factor = 1 start_manpower_factor = 1"
			owner = RZD
			count = 5
		}
	}
}

projectz_rzd_spread_war_to_neighbors = {
	limit = {
		num_of_controlled_states > 0
	}

	every_neighbor_country = {
		limit = {
			exists = yes
			NOT = { tag = RZD }
			NOT = { has_war_with = RZD }
		}

		save_event_target_as = projectz_rzd_neighbor_target

		RZD = {
			if = {
				limit = {
					NOT = { has_wargoal_against = event_target:projectz_rzd_neighbor_target }
				}
				create_wargoal = {
					type = annex_everything
					target = event_target:projectz_rzd_neighbor_target
					expire = 0
				}
			}
			declare_war_on = {
				target = event_target:projectz_rzd_neighbor_target
				type = annex_everything
			}
		}
	}
}

projectz_rzd_enforce_isolation = {
	if = {
		limit = {
			is_in_faction = yes
		}
		leave_faction = yes
	}

	every_other_country = {
		limit = {
			exists = yes
			NOT = { tag = RZD }
		}

		diplomatic_relation = {
			country = RZD
			relation = non_aggression_pact
			active = no
		}

		diplomatic_relation = {
			country = RZD
			relation = military_access
			active = no
		}

		diplomatic_relation = {
			country = RZD
			relation = guarantee
			active = no
		}

		diplomatic_relation = {
			country = RZD
			relation = docking_rights
			active = no
		}

		diplomatic_relation = {
			country = RZD
			relation = embargo
			active = yes
		}
	}
}
